## County-Level Data Statistics:

Statistics of CTAmap county-level data of each type:

```matlab
# -*- coding: utf-8 -*-
import arcpy
import collections

# Set workspace
workspace = r"F:\shengshxiain.gdb"
arcpy.env.workspace = workspace

# Feature class name
feature_class = "T2024 Early County-level"

# Define list of county-level types
county_types = ['District', 'County-level City', 'County', 'Autonomous County', 'Banner',  'Forestry Area', 'Special District','Not Counted']

# Create a nested defaultdict to store statistics results
stats = collections.defaultdict(lambda: collections.defaultdict(int))

# Use SearchCursor to traverse the feature class
with arcpy.da.SearchCursor(feature_class, ["Province", "County-level Type"]) as cursor:
    for row in cursor:
        province = row[0]
        county_type = row[1]
        stats[province][county_type] += 1

# Print statistics results
print("County-level Type Statistics for each Province-level Unit:")
# Print table header
header = "Province"
for type_name in county_types:
    header += f"\t{type_name}"
header += "\tTotal"
print(header)

# Print data for each province
for province, types in stats.items():
    # Calculate total excluding "Not Counted"
    total = sum(count for type_name, count in types.items() if type_name != 'Not Counted')
    line = province
    for type_name in county_types:
        line += f"\t{types[type_name]}"
    line += f"\t{total}"
    print(line)

# Calculate totals
total_stats = collections.defaultdict(int)
for types in stats.values():
    for county_type, count in types.items():
        total_stats[county_type] += count

# Calculate total excluding "Not Counted"
total_sum = sum(count for type_name, count in total_stats.items() if type_name != 'Not Counted')

print("\nTotals:")
for type_name in county_types:
    print(f"{type_name}: {total_stats[type_name]}")
print(f"Total: {total_sum}")
```
